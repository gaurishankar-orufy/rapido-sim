  async function api(path, options) {
    const res = await fetch(API + path, { headers: { 'Content-Type': 'application/json' }, ...options });
    const body = await res.json().catch(() => ({}));
    if (!res.ok && res.status !== 409) throw new Error(body.reason || 'HTTP ' + res.status);
    return body;
  }

  $('book').onclick = async () => {
    $('bookErr').textContent = '';
    let data;
    try { data = JSON.parse($('data').value || '{}'); } catch { return ($('bookErr').textContent = 'Data is not valid JSON'); }
    const drivers = $('drivers').value.split(/[\s,]+/).map((d) => d.trim()).filter(Boolean);
    $('book').disabled = true;
    try {
      active = await api('/api/rides', { method: 'POST', body: JSON.stringify({

    $('book').disabled = false;

  const label = { searching: 'Finding a driver…', accepted: 'Driver found 🎉', no_drivers: 'No drivers found', cancelled: 'Ride cancelled', failed: 'Send failed' };
  const latest = (r) => r.attempts[r.attempts.length - 1] || {};
  const offeredTo = (r) => (latest(r).to?.length ? latest(r).to.join(', ') : 'everyone');

  function render() {
    if (!active) return;
    $('activeCard').hidden = false;
    $('activeId').textContent = active.rideId;
    $('activeStatus').className = 'status ' + active.status;
    let text = label[active.status] || active.status;
    if (active.status === 'searching') text += ` (offered to ${offeredTo(active)})`;
    if (active.status === 'searching' && active.expiresAt) {
      const left = Math.max(0, Math.round((new Date(active.expiresAt) - Date.now()) / 1000));
      text += ` ${Math.floor(left / 60)}:${String(left % 60).padStart(2, '0')}`;
    }
    if (active.status === 'accepted') text += ` · ${active.acceptedBy}`;
    $('activeStatus').textContent = text;
    $('activeMeta').textContent = `${active.mode === 'sequential' ? 'one by one' : 'all at once'} · request ${latest(active).requestId || '—'} · ` +
      `offers ${active.attempts.length}${active.mode === 'sequential' ? '/' + active.drivers.length : ''} · rejected by ${active.rejectedBy.length}`;
    $('activeEvents').textContent = active.events.map((e) => `${e.at.slice(11, 19)}  ${e.type}${e.detail ? '  ' + JSON.stringify(e.detail) : ''}`).join('\n');
  }

  async function poll() {
    try {
      const list = await api('/api/rides');
      $('rides').innerHTML = list.map((r) => `${esc(r.rideId)}${esc(r.screenId)}${esc(r.mode)}` +
        `${esc(r.status)}${esc(offeredTo(r))}${esc(r.acceptedBy)}` +
        `${r.rejectedBy.length}${esc(latest(r).requestId)}`).join('');
      if (active) { active = list.find((r) => r.rideId === active.rideId) || active; render(); }
    } catch { /* server asleep or restarting: try again next tick */ }
  }

  async function loadLogs() {
    try {
      const logs = await api('/api/logs');
      $('logs').textContent = logs.length ? JSON.stringify(logs, null, 2) : 'No calls yet.';
    } catch (e) { $('logs').textContent = e.message; }
  }

  setInterval(poll, 2000);
  setInterval(() => active?.status === 'searching' && render(), 1000);
  poll();
